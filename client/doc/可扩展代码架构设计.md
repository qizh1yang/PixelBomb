## 1. 设计原则

### 1.1 核心原则

| 原则 | 说明 | 应用场景 |
|------|------|----------|
| **数据驱动** | 使用 Godot Resource 定义角色、地图、道具的属性 | 添加新角色只需创建 `.tres` 文件 |
| **工厂模式** | 使用工厂类创建角色、地图等对象 | 统一创建逻辑，易于扩展 |
| **注册表模式** | 使用注册表管理所有可用角色、地图等 | 动态获取可用列表，无需硬编码 |
| **信号驱动** | 使用 Godot 信号进行松耦合通信 | UI 与逻辑分离，易于修改 |
| **MVC 分离** | Model（数据）- View（UI）- Controller（逻辑）分离 | 修改 UI 不影响游戏逻辑 |

### 1.2 扩展性目标

- ✅ **添加新角色**：只需创建 `CharacterResource` + 角色场景，无需修改现有代码
- ✅ **添加新地图**：只需创建 `MapResource` + 地图生成器，自动出现在地图选择列表
- ✅ **修改 UI**：UI 与逻辑分离，修改 UI 场景不影响后台逻辑
- ✅ **添加新游戏模式**：基于现有架构，添加新模式只需新增房间设置

## 3. 角色系统架构

### 3.1 设计目标

- 添加新角色只需：
  1. 创建 `CharacterResource` (`.tres`)
  2. 创建角色场景 (`.tscn`)
  3. 在 `CharacterRegistry` 中注册（自动扫描或手动注册）
- 无需修改现有代码

### 3.2 核心类设计

#### CharacterResource (Resource)

```gdscript
# resources/characters/character_base.gd
extends Resource
class_name CharacterResource

@export var character_id: String       # 唯一 ID (如 "fire", "ice")
@export var display_name: String       # 显示名称 (如 "烈焰破壁者")
@export var description: String        # 角色描述
@export var icon: Texture2D           # 角色图标
@export var preview_scene: PackedScene # 预览场景 (用于角色选择界面)

# ── 基础属性 (可被道具提升) ──
@export var base_speed: float = 100.0    # 基础移动速度
@export var base_bomb_cap: int = 1       # 基础炸弹数量
@export var base_radius: int = 1         # 基础爆炸范围
@export var base_shield_count: int = 0   # 基础护盾数量

# ── 属性天花板 (最高可提升至多少) ──
@export var max_speed: float = 200.0      # 最高速度
@export var max_bomb_cap: int = 5        # 最多炸弹数
@export var max_radius: int = 8          # 最大爆炸范围
@export var max_shield_count: int = 3    # 最多护盾数

# ── 特殊能力 (可选，用于后续扩展) ──
@export var special_ability: String = ""  # 特殊能力 ID (如 "double_bomb")
@export var ability_description: String = "" # 能力描述
```

#### CharacterFactory (工厂类)

```gdscript
# autoloads/character_factory.gd
extends Node
class_name CharacterFactory

# 根据角色 ID 创建角色实例
func create_character(character_id: String) -> PlayerController:
    var char_res: CharacterResource = CharacterRegistry.get_character(character_id)
    if not char_res:
        push_error("Character not found: " + character_id)
        return null
    
    # 加载角色场景
    var scene_path = "res://prefabs/Players/player%d/player%d.tscn" % [char_res.char_index + 1, char_res.char_index + 1]
    var scene = load(scene_path)
    
    var instance = scene.instantiate()
    
    # 应用角色属性
    instance.currentSpeed = char_res.base_speed
    instance.currentMaxBombs = char_res.base_bomb_cap
    instance.currentRadius = char_res.base_radius
    instance.currentShields = char_res.base_shield_count
    
    # 设置天花板
    instance.speedCap = char_res.max_speed
    instance.maxBombsCap = char_res.max_bomb_cap
    instance.explosionRadiusCap = char_res.max_radius
    instance.maxShieldsCap = char_res.max_shield_count
    
    return instance
```

#### CharacterRegistry (注册表)

```gdscript
# autoloads/character_registry.gd
extends Node
class_name CharacterRegistry

var characters: Dictionary = {}  # character_id -> CharacterResource

func _ready() -> void:
    _scan_characters()

# 扫描所有角色资源并注册
func _scan_characters() -> void:
    var dir = DirAccess.open("res://resources/characters/")
    if not dir:
        return
    
    dir.list_dir_begin()
    var file_name = dir.get_next()
    while file_name != "":
        if file_name.ends_with(".tres"):
            var res_path = "res://resources/characters/" + file_name
            var char_res = load(res_path) as CharacterResource
            if char_res:
                register_character(char_res)
        file_name = dir.get_next()

func register_character(char_res: CharacterResource) -> void:
    characters[char_res.character_id] = char_res

func get_character(character_id: String) -> CharacterResource:
    return characters.get(character_id)

func get_all_characters() -> Array:
    return characters.values()
```

2. **创建角色场景** (`player5/player5.tscn`)：
   - 基于 `player_controller.gd`
   - 替换精灵图

3. **完成**！角色会自动被 `CharacterRegistry` 扫描并注册

```gdscript
# resources/maps/generators/classic_map_generator.gd
extends MapGenerator
class_name ClassicMapGenerator

func generate(map_res: MapResource, seed: int = 0) -> Array:
    if seed != 0:
        seed = seed
    
    var map = []
    for x in range(map_res.width):
        map.append([])
        for y in range(map_res.height):
            # 边界墙
            if x == 0 or x == map_res.width - 1 or y == 0 or y == map_res.height - 1:
                map[x].append(1)
            # 支柱 (偶数坐标)
            elif x % 2 == 0 and y % 2 == 0:
                map[x].append(1)
            # 随机可破坏墙
            else:
                map[x].append(2 if randf() < map_res.wall_density else 0)
    
    # 清理出生点
    _clear_spawn_zones(map, map_res)
    
    return map

func _clear_spawn_zones(map: Array, map_res: MapResource) -> void:
    var safe_size = map_res.safe_zone_size
    # 左上角
    for x in range(1, safe_size + 1):
        for y in range(1, safe_size + 1):
            map[x][y] = 0
    # 其他角落...
```

## 5. UI系统架构

### 5.1 设计目标

- UI 与逻辑分离（MVC 模式）
- 统一的 UI 管理（导航、转场动画）
- 易于添加新的 UI 界面
- 支持 UI 主题切换（后续扩展）

### 5.2 核心类设计

#### UIManager (UI 管理器)

```gdscript
# autoloads/ui_manager.gd
extends Node
class_name UIManager

signal screen_changed(old_screen: String, new_screen: String)

var current_screen: String = ""
var screen_cache: Dictionary = {}  # screen_name -> Node

# UI 场景路径映射
const SCREEN_PATHS = {
    "login": "res://scenes/Login/login.tscn",
    "main_menu": "res://scenes/MainMenu/main_menu.tscn",
    "lobby": "res://scenes/Lobby/lobby.tscn",
    "room": "res://scenes/Room/room.tscn",
    "game": "res://stages/game_stage.tscn",
    "settlement": "res://scenes/Settlement/settlement.tscn",
}

func _ready() -> void:
    # 预加载常用 UI
    _preload_screen("login")

# 切换到指定界面
func change_screen(screen_name: String, use_transition: bool = true) -> void:
    if not SCREEN_PATHS.has(screen_name):
        push_error("Unknown screen: " + screen_name)
        return
    
    var old_screen = current_screen
    
    if use_transition:
        await _play_transition_out()
    
    # 卸载当前界面
    if old_screen != "":
        _unload_screen(old_screen)
    
    # 加载新界面
    var screen = _load_screen(screen_name)
    get_tree().root.add_child(screen)
    
    current_screen = screen_name
    screen_changed.emit(old_screen, current_screen)
    
    if use_transition:
        await _play_transition_in()

# 加载界面 (带缓存)
func _load_screen(screen_name: String) -> Node:
    if screen_cache.has(screen_name):
        return screen_cache[screen_name]
    
    var path = SCREEN_PATHS[screen_name]
    var scene = load(path)
    var instance = scene.instantiate()
    
    screen_cache[screen_name] = instance
    return instance

# 卸载界面
func _unload_screen(screen_name: String) -> void:
    if screen_cache.has(screen_name):
        var screen = screen_cache[screen_name]
        if screen.is_inside_tree():
            screen.queue_free()
        screen_cache.erase(screen_name)
```

#### BaseScreen (界面基类)

```gdscript
# scenes/BaseScreen/base_screen.gd
extends Control

# 界面 ID (用于 UIManager 识别)

# 初始化界面

    # 播放进入动画
    _play_enter_animation()

# 播放进入动画
func _play_enter_animation() -> void:
    # 默认：淡入
    modulate.a = 0
    var tween = create_tween()
    tween.tween_property(self, "modulate:a", 1.0, 0.3)

# 播放退出动画
func _play_exit_animation() -> void:
    var tween = create_tween()
    tween.tween_property(self, "modulate:a", 0.0, 0.3)
    await tween.finished
```

```gdscript
            "map_id": settings.map_id,
            "game_duration": settings.game_duration,
        }
    })
    settings_changed.emit()
```

```gdscript
        "player_id": my_id,
        "player_name": PlayerData.player_name
    })
    
    http_request.request(url, headers, HTTPClient.METHOD_POST, body)
    http_request.request_completed.connect(_on_room_created, CONNECT_ONE_SHOT)

# 加入房间
func join_room(room_id: String) -> void:
    var url = "http://localhost:8080/join_room"
    var headers = ["Content-Type: application/json"]
    var body = JSON.stringify({
        "room_id": room_id,
        "player_id": my_id,
        "player_name": PlayerData.player_name
    })
    
    http_request.request(url, headers, HTTPClient.METHOD_POST, body)
    http_request.request_completed.connect(_on_room_joined, CONNECT_ONE_SHOT)

# 发送数据 (WebSocket)
func send_data(data: Dictionary) -> void:
    if ws_peer and ws_peer.get_ready_state() == WebSocketPeer.STATE_OPEN:
        var json_str = JSON.stringify(data)
        ws_peer.put_packet(json_str.to_utf8_buffer())
```
