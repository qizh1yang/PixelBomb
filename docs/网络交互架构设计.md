## 2. 核心原则

| 原则 | 说明 |
|------|------|
| **服务器权威** | 所有游戏逻辑（移动验证、伤害计算、胜负判定）在 Go 后端执行，客户端只做表现层 |
| **状态同步** | 客户端发送操作指令，服务器广播权威游戏状态 |
| **分层通信** | 大厅用 HTTP 短连接，对局用 WebSocket 长连接 |
| **并发安全** | 房间状态用 `sync.Mutex` 保护，或用 channel 传递消息 |
| **断线重连** | 玩家断开后保留状态 30 秒，允许携带 token 重连 |

### 5.3 游戏 Tick 循环

```go
func (r *Room) StartGameLoop() {
    // 1. 初始化游戏状态
    r.GameState = &GameState{...}
    
    // 2. 初始化地图、玩家、宝箱
    r.initMap()
    r.initPlayers()
    r.initChests()
    
    // 3. 广播游戏开始
    r.broadcastGameStart()
    
    // 4. 主循环 (20 Tick/s = 50ms)
    ticker := time.NewTicker(50 * time.Millisecond)
    defer ticker.Stop()
    
    for range ticker.C {
        r.lock.Lock()
        if !r.inGame || r.GameState.GameOver {
            r.lock.Unlock()
            break
        }
        r.lock.Unlock()
        
        r.updateGame()
    }
}

func (r *Room) updateGame() {
    gs := r.GameState
    now := time.Now()
    elapsed := now.Sub(gs.StartTime).Seconds()
    
    // 1. 检查炸弹爆炸
    r.checkBombs(now)
    
    // 2. 检查玩家碰撞（道具、撤离点）
    r.checkPlayerCollisions()
    
    // 3. 激活撤离（3分钟）
    if !gs.EvacActivated && elapsed >= 180 {
        gs.EvacActivated = true
        r.broadcastMsg("evac_activated", nil)
    }
    
    // 4. 游戏结束检查（5分钟）
    if elapsed >= 300 {
        r.endGame("timeout")
    }
    
    // 5. 广播游戏状态（可选，用于状态同步）
    r.broadcastGameState()
}
```

### 6.3 WebSocket 对局通信

```gdscript
# 连接到对局服务器
func connect_to_game(ws_url: String, token: String = "") -> void:
    state = ConnectionState.CONNECTING
    
    ws_peer = WebSocketPeer.new()
    var full_url = ws_url
    if token != "":
        full_url += "?token=" + token
    
    ws_peer.connect_to_url(full_url)
    _set_process(true)

# 发送数据
func send_data(data: Dictionary) -> void:
    if ws_peer and ws_peer.get_ready_state() == WebSocketPeer.STATE_OPEN:
        var json_str = JSON.stringify(data)
        ws_peer.put_packet(json_str.to_utf8_buffer())

# 处理 WebSocket 消息
func _process(_delta: float) -> void:
    if not ws_peer:
        return
    
    ws_peer.poll()
    
    var state = ws_peer.get_ready_state()
    if state == WebSocketPeer.STATE_OPEN:
        # 接收消息
        while ws_peer.get_available_packet_count():
            var packet = ws_peer.get_packet()
            var msg = JSON.parse_string(packet.get_string_from_utf8())
            _handle_message(msg)
    elif state == WebSocketPeer.STATE_CLOSED:
        # 连接关闭，触发重连
        _on_connection_closed()

# 处理接收到的消息
func _handle_message(msg: Dictionary) -> void:
    match msg.get("type"):
        "ROOM_STATE":
            room_players = msg.get("players", [])
            emit_signal("room_state_updated", room_players)
        
        "GAME_START":
            emit_signal("game_started")
        
        "PLAYER_MOVE":
            # 更新其他玩家的位置
            _update_player_position(msg.payload)
        
        "BOMB_PLACED":
            # 创建炸弹表现
            _spawn_bomb_visual(msg.payload)
        
        "GAME_OVER":
            # 游戏结束，返回房间
            _on_game_over(msg.payload)
        
        "ERROR":
            emit_signal("error_received", msg.get("message", "未知错误"))
```

func handleCreateRoom(w http.ResponseWriter, r *http.Request) {

## 9. 重连机制

### 9.1 Token 设计

使用 JWT (JSON Web Token) 或简单签名 Token：

```
Token 结构：
{
  "player_id": "player_123456",
  "room_id": "ABC123",
  "exp": 1620000000  // 过期时间（可选）

```

### 9.2 重连流程

```
客户端                            服务端
  |                                   |
  |  (网络断开)                        |
  |                                   |
  |---- 1. 检测断开 ------------------>|
  |     启动重连计时器                  |
  |                                   |
  |---- 2. 重新 WebSocket 连接 ------>|
  |     (携带 token)                   |
  |                                   |
  |<---- 3. 验证 token --------------|
  |     恢复玩家状态                    |
  |                                   |
  |<---- 4. 发送当前游戏状态 --------|
  |     (如果游戏中)                    |
  |                                   |
  |---- 5. 继续游戏 ----------------->|
  |                                   |
```

### 9.3 服务端重连处理

```go
// 在 WebSocket 连接时检查 token
func serveWs(hub *Hub, w http.ResponseWriter, r *http.Request) {
    token := r.URL.Query().Get("token")
    
    var clientID string
    var roomID string
    
    if token != "" {
        // 验证 token，尝试恢复连接
        clientID, roomID = validateToken(token)
    }
    
    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil {
        return
    }
    
    client := &Client{
        hub:      hub,
        conn:     conn,
        send:     make(chan []byte, 256),
        ID:       clientID,
        room:     nil,
    }
    
    client.hub.register <- client
    
    // 如果 token 有效，自动重新加入房间
    if clientID != "" && roomID != "" {
        client.hub.joinRoom(client, roomID)
    }
    
    go client.writePump()
    go client.readPump()
}
```

### 9.4 客户端重连逻辑

```gdscript
# NetworkManager.gd
var reconnect_timer: Timer
var max_reconnect_attempts: int = 5
var reconnect_attempts: int = 0

func _on_connection_closed() -> void:
    state = ConnectionState.RECONNECTING
    reconnect_attempts = 0
    
    if reconnect_timer == null:
        reconnect_timer = Timer.new()
        reconnect_timer.one_shot = false
        reconnect_timer.timeout.connect(_attempt_reconnect)
        add_child(reconnect_timer)
    
    reconnect_timer.start(2.0)  # 每 2 秒尝试重连

func _attempt_reconnect() -> void:
    reconnect_attempts += 1
    
    if reconnect_attempts > max_reconnect_attempts:
        reconnect_timer.stop()
        state = ConnectionState.DISCONNECTED
        emit_signal("error_received", "重连失败，请重新加入房间")
        return
    
    # 尝试重新连接
    if reconnect_token != "" and reconnect_room != "":
        connect_to_game(ws_url, reconnect_token)
    else:
        # 无 token，需要重新加入房间
        join_room(reconnect_room)
```
